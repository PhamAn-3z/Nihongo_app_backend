Table Membership {
  id serial [pk]
  Duration "character varying(50)" [not null] 
  membershipRank "character varying(50)" [not null] 
  price "double precision" [not null]
  maxFlashcardSet integer [not null]
  isActive boolean [default: true]
}

Table PromoCode {
  id serial [pk]
  sales "double precision" [not null]
  Expired boolean [default: false]
  dayCreated date [default: `now()`]
  dayExpired date [not null]
}

Table Receipt {
  id serial [pk]
  user_id integer [not null] 
  membershipId integer [not null]
  promoCodeId integer
  paymentMethod "character varying(100)" [not null]
  total "double precision" [not null, default: 0]
  is_paid boolean [not null, default: false] 
  dayCreated date [default: `now()`]
}

Table user_stats {
  user_id integer [pk] 
  current_streak integer [default: 0]
  max_streak integer [default: 0]
  total_exp integer [default: 0]
  membership_id integer [default: 1] 
  isActive boolean [default: true]     
  last_study_date date
}
